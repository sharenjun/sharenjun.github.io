指针的简单应用

#include <iostream>
#include <stdio.h>
#include <malloc.h>

int main()
{
    int x;
    int *q,*p;
    //p default set scanf x value;
    //q is default set for malloc;
    x=10;
    p=&x;
    q=(int *)malloc(100);
    *q=100;
    printf("*p=%d,*q=%d\n",*p,*q);
    free(q);
    return 0;
}

{
    int t;

    *x = *y;
    *y = t;
